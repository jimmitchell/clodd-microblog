{{- /* theme.css is one file, split at its marker: everything above is inlined so first paint needs no request, everything below arrives as a preloaded stylesheet. Sections below the marker keep their order relative to those above, so the split never changes which rule wins. The inlined half resolves url()s against the page, not a stylesheet, so the font paths are rewritten to site-relative URLs here. That keeps a site served from a subpath working. */ -}} {{- $marker := "/* =END CRITICAL= */" -}} {{- $parts := split (resources.Get "css/theme.css").Content $marker -}} {{- $fonts := "fonts/" | relURL -}} {{- $critical := replace (index $parts 0) "url('/fonts/" (printf "url('%s" $fonts) -}} {{- /* The external-link arrow excludes links back to the site itself, which Clodd's stylesheet spells out as its own host. */ -}} {{- $critical = replace $critical `[href*="//jimmitchell.org"]` (printf `[href*="//%s"]` (urls.Parse site.BaseURL).Host) -}} {{- $critical = resources.FromString "css/theme.critical.css" $critical | minify -}} {{- /* clodd-hugo.css styles the pages Clodd does not have; it joins the deferred half, after Clodd's own rules. */ -}} {{- $extras := (resources.Get "css/clodd-hugo.css").Content -}} {{- $deferred := resources.FromString "css/theme.deferred.css" (printf "%s\n%s" (index $parts 1) $extras) | minify | fingerprint -}}
