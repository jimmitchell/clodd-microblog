{{- /* Feed discovery. Every page advertises the site's feeds; a category or tag archive also advertises its own, titled with the term so a reader can tell them apart. Any output format with a feed media type counts, so a site's own feed formats are picked up as they are defined. One link per feed type: a site that serves the same feed at a second URL advertises only the first. Written with printf because html/template would otherwise escape the + in each media type as +, which a naive feed finder doesn't decode. */ -}} {{- $feedTypes := slice "application/rss+xml" "application/atom+xml" "application/feed+json" -}} {{- $seen := slice }} {{- range site.Home.AlternativeOutputFormats }} {{- if and (in $feedTypes .MediaType.Type) (not (in $seen .MediaType.Type)) }} {{- $seen = $seen | append .MediaType.Type }} {{ printf `` .MediaType.Type (site.Title | htmlEscape) .Permalink | safeHTML }} {{- end }} {{- end }} {{- if eq .Kind "term" }} {{- $title := printf "%s — %s: %s" site.Title (.Data.Singular | humanize) .Title }} {{- $seen = slice }} {{- range .AlternativeOutputFormats }} {{- if and (in $feedTypes .MediaType.Type) (not (in $seen .MediaType.Type)) }} {{- $seen = $seen | append .MediaType.Type }} {{ printf `` .MediaType.Type ($title | htmlEscape) .Permalink | safeHTML }} {{- end }} {{- end }} {{- end }}
