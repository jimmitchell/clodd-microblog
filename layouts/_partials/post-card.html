{{- if eq $kind "article" }}

{{ $p.Title }}

{{- end }} {{- if eq $kind "note" }} {{- /* Notes have no title, so the date carries the permalink. A photo post's date moves below the picture instead. */}} {{ $date }} {{- end }} {{- with $contexts }}
{{ . }}
{{- end }} {{- if eq $kind "article" }} {{- /* A titled post's image illustrates rather than leads: a thumbnail. u-featured, since an article's lead picture is not its photo property. {{ $body | safeHTML }}
