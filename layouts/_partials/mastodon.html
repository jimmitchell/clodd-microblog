{{- /* params.social.mastodon, which may be written as a handle (@user@instance) or a profile URL (https://instance/@user), resolved to both forms. Empty strings when unset or not recognisably either. */ -}} {{- $out := dict "url" "" "handle" "" -}} {{- with (site.Params.social | default dict).mastodon -}} {{- with findRESubmatch `^https://([^/]+)/@([^/@]+)/?$` . 1 -}} {{- $m := index . 0 -}} {{- $out = dict "url" (index $m 0) "handle" (printf "@%s@%s" (index $m 2) (index $m 1)) -}} {{- else -}} {{- with findRESubmatch `^@?([^@\s]+)@([^@\s]+)$` . 1 -}} {{- $m := index . 0 -}} {{- $out = dict "url" (printf "https://%s/@%s" (index $m 2) (index $m 1)) "handle" (printf "@%s@%s" (index $m 1) (index $m 2)) -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
