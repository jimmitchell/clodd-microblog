{{- /* The context lines above a reply, like, repost, bookmark or RSVP: "↩ In reply to example.com/…", marked up so the URL is u-in-reply-to and so on. Clodd's order: repost, like, reply, bookmark. Front matter in_reply_to, like_of, repost_of and bookmark_of each take a URL or a list of them. An rsvp (yes, no, maybe, interested) relabels the first in_reply_to only: an RSVP answers one event, and a second reply target stays an ordinary reply. The value goes in a so a parser reads "yes" rather than the words. Renders nothing when there are no contexts. */ -}} {{- $p := . -}} {{- $kinds := slice (dict "key" "repost_of" "symbol" "♺" "verb" "Reposted" "class" "u-repost-of") (dict "key" "like_of" "symbol" "♥" "verb" "Liked" "class" "u-like-of") (dict "key" "in_reply_to" "symbol" "↩" "verb" "In reply to" "class" "u-in-reply-to") (dict "key" "bookmark_of" "symbol" "" "verb" "Bookmarked" "class" "u-bookmark-of") -}} {{- $rsvpLabels := dict "yes" (slice "✓" "Going to") "no" (slice "✗" "Not going to") "maybe" (slice "?" "Maybe going to") "interested" (slice "☆" "Interested in") -}} {{- $rsvp := lower (strings.TrimSpace (string ($p.Params.rsvp | default ""))) -}} {{- $rsvpLeft := isset $rsvpLabels $rsvp -}} {{- range $kinds -}} {{- $kind := . -}} {{- $urls := index $p.Params .key | default slice -}} {{- if reflect.IsSlice $urls | not }}{{ $urls = slice $urls }}{{ end -}} {{- range $urls -}} {{- $url := strings.TrimSpace (string .) -}} {{- if $url -}} {{- $symbol := $kind.symbol -}} {{- $verb := $kind.verb -}} {{- $answer := "" -}} {{- if and (eq $kind.key "in_reply_to") $rsvpLeft -}} {{- $symbol = index (index $rsvpLabels $rsvp) 0 -}} {{- $verb = index (index $rsvpLabels $rsvp) 1 -}} {{- $answer = $rsvp -}} {{- $rsvpLeft = false -}} {{- end -}} {{- $text := replaceRE `^https?://` "" $url -}} {{- if gt (len $text) 60 }}{{ $text = printf "%s…" (substr $text 0 57) }}{{ end }}
