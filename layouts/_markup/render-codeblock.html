{{- /* A fenced code block, as Clodd's HighlightFencedCodeRenderer writes one: the .code-block wrapper the copy button anchors to (emitted here so theme.js doesn't reflow the page by adding it), and a dark xcode-dark 
 in light mode and dark alike.

       Chroma highlights it with classes rather than inline styles; inline
       mode returns only the token spans, which go into Clodd's own .
       clodd-hugo.css colours Chroma's classes with Clodd's palette. A block
       with no language, or one Chroma doesn't know, is plain text in the
       same dark block. */ -}}
{{- $lang := .Type | default "" -}}
{{- $inner := "" -}}
{{- if $lang -}}
  {{- $h := transform.Highlight .Inner $lang (dict "noClasses" false "hl_inline" true) -}}
  {{- $inner = replaceRE `^]*>|$` "" (string $h) -}}
{{- else -}}
  {{- $inner = .Inner | htmlEscape -}}
{{- end -}}

{{ $inner | safeHTML }}
